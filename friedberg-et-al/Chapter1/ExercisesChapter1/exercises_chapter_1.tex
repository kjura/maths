\begin{theorem}
    Corollary 1
\end{theorem}
\begin{proof}
    Assume we are working in a vector space \(V\) over a field \(F\). Assume for contradiction,
that \(0 \in V\) is not unique. Then there exists two zero vectors \(0_{1} \neq 0_{2}\).
It follows by (VS 3) and (VS 1):

\begin{equation*}
    0_{1} = 0_{1} + 0_{2} = 0_{2} + 0_{1} = 0_{2}
\end{equation*}

but \(0_{1} \neq 0_{2}\), contradiction.
\end{proof}

\begin{theorem}
    Corollary 2
\end{theorem}
\begin{proof}
    Assume a vector space \(V\) over a field \(F\). Suppose \(x \in V\) and assume for contradiction
    that there exists two vectors in \(V\), \(y_1 \neq y_2\) such that:
    \begin{equation*}
        x + y_1 = 0 \ \ \mbox{and} \ \ x + y_2 = 0
    \end{equation*}
    It follows then:
    \begin{equation*}
        x + y_1 = 0 = x + y_2
    \end{equation*}
    And using theorem 1.2.1 (Cancellation Law For Vector Addition):
    \begin{equation*}
        y_1 = y_2
    \end{equation*}
    Contradiction.
\end{proof}

\section*{Exercise 1.2.11}

Let \(V = {0}\) consist of a single vector \(0\) and define \(0 + 0 = 0\) and
\(c0 = 0\) for each scalar \(c \in F\). We will show that \(V\) is a vector
space over \(F\).

Suppose \(x, y \in V\). Since \(V\) has only one element
that is a single vector \(0\), \(x = y = 0\).

Suppose that there exists two vectors \(0_{1}, 0_{2} \in V\) satisfying

\begin{equation*}
    x + y = 0_{1} \ \ \mbox{and} \ \ x + y = 0_{2}
\end{equation*}

From the definition of addition it follows that:

\begin{equation*}
    0_{1} = x + y = 0 + 0 = 0 
\end{equation*}
And also
\begin{equation*}
    0_{2} = x + y = 0 + 0 = 0
\end{equation*}

It follows:

\begin{equation*}
    0_{1} = 0_{2} = 0
\end{equation*}

and shows that for every pair of elements \(x, y \in V\) there is a unique element \(x + y\) also
in \(V\). 


Suppose now that \(c \in F\) and \(x \in V\). Since \(V\) has only one element
that is a single vector \(0\), \(x = 0\). Suppose that there exists two vectors \(a, b \in V\) satisfying

\begin{equation*}
    cx = a \ \ \mbox{and} \ \ cx = b
\end{equation*}

It follows:

\begin{equation*}
    cx = c \cdot 0 = 0 = a
\end{equation*}
\begin{equation*}
    cx = c \cdot 0 = 0 = b
\end{equation*}

So:

\begin{equation*}
    a = 0 = b
\end{equation*}

and shows that for every \(x \in V\) there is a unique element \(cx\) also
in \(V\). We are now ready to check 8 conditions necessary for concluding if \(V\) is a vector space.

Suppose \(x, y \in V\). Since \(V\) has only one element
that is a single vector \(0\), \(x = y = 0\).  We have:

\begin{equation*}
    x + y = 0 + 0 = 0
\end{equation*}
but also:

\begin{equation*}
    y + x = 0 + 0 = 0
\end{equation*}

It follows:

\begin{equation*}
    x + y = y + x
\end{equation*}

\subsubsection*{(VS 2)}

Suppose \(x, y, z \in V\). Since \(V\) has only one element
that is a single vector \(0\), \(x = y = z = 0\). We have:

\begin{equation*}
    (x + y) + z = (0 + 0) + 0 = 0 + 0 = 0
\end{equation*}
\begin{equation*}
    x + (y + z) = 0 + (0 + 0) = 0 + 0 = 0
\end{equation*}

It follows:

\begin{equation*}
    (x + y) + z = x + (y + z)
\end{equation*}

\subsubsection*{(VS 3)}

Let \(\overline{0}\) be some element in \(V\). Suppose \(x \in V\). Since \(V\) has only one element
that is a single vector \(0\), \(x = \overline{0} = 0\). It follows:

\begin{equation*}
    x + \overline{0} = x + x = 0 + 0 = 0 = x
\end{equation*}

\subsubsection*{(VS 4)}

Suppose \(x \in V\). Let \(y\) be some element in \(V\). Notice that since \(V\) has only one element
\(x = y = 0\) and recall \(\overline{0} = 0\) It follows:

\begin{equation*}
    x + y 
\end{equation*}
\begin{equation*}

\end{equation*}
\begin{equation*}
    = \overline{0} + \overline{0} \ \ \ \emph{Since \(\overline{0} = 0\)}
\end{equation*}
\begin{equation*}
    = \overline{0} \ \ \ \emph{Definition of addition on \(V\)}
\end{equation*}

\subsubsection*{(VS 5)}

Suppose \(x \in V\). Notice that since \(V\) has only one element
\(x = 0\). Let \(\overline{1} \in F\). Recall that \(c0 = 0\) for each scalar \(c \in F\).
We have:

\begin{equation*}
    \overline{1}x = \overline{1} \cdot 0 = 0 = x
\end{equation*}

\subsubsection*{(VS 6)}

Suppose \(a, b \in F\) and \(x \in V\). We have:

\begin{equation*}
    (ab)x = (ab)0 = 0 \ \ \ \emph{Definition of scalar multiplication}
\end{equation*}
and also
\begin{equation*}
    a(bx) = a(b0) = a \cdot 0 = 0 \ \ \ \emph{Definition of scalar multiplication}
\end{equation*}

It follows:

\begin{equation*}
    (ab)x = a(bx)
\end{equation*}

Suppose \(a \in F\) and \(x, y \in V\) and notice \(x = y = 0\). We have:

\begin{equation*}
    a(x + y) = 
\end{equation*}
\begin{equation*}

\end{equation*}
\begin{equation*}
    = 0 \ \ \ \emph{Definition of scalar multiplication in V}
\end{equation*}
\begin{equation*}
    = 0 + 0 \ \ \ \emph{Definition of addition in V}
\end{equation*}
\begin{equation*}

\end{equation*}

Suppose \(a, b \in F\) and \(x \in V\). Since \(F\) is a field, \(a + b\) is also in \(F\).

\section*{Exercise 1.2.12}

Let \(\mathcal{F}(\mathbb{R}, \mathbb{R})\) be the set
of even functions defined on the real line with the following
definitions for the operations of addition and scalar multiplication:

\begin{equation*}
    (f + g)(s) = f(s) + g(s) \ \mbox{and} \ (cf)(s) = c[f(s)]
\end{equation*}

\subsection*{Addition and scalar multiplication are well-defined operations}

First, let us show that the operations defined above are well-defined. Suppose \(f, g \in \mathcal{F}(\mathbb{R}, \mathbb{R})\).
Suppose also that \(c, s \in \mathbb{R}\). Notice that \((-s) \in \mathbb{R}\) as well.
We have for addition:
\begin{align*}
    (f + g)(s) &= f(s) + g(s) && \text{Definition of addition in \(\mathcal{F}(\mathbb{R}, \mathbb{R})\)} \\
               &= f(-s) + g(-s) && \text{\(f, g\) are even} \\
               &= (f + g)(-s) && \text{Definition of addition in \(\mathcal{F}(\mathbb{R}, \mathbb{R})\)}
\end{align*}
And for scalar multiplication:
\begin{align*}
    (cf)(s) &= c\Big[f(s)\Big] && \text{\text{Definition of scalar multiplication in \(\mathcal{F}(\mathbb{R}, \mathbb{R})\)}} \\
            &= c\Big[f(-s)\Big] && \text{\(f\) is even} \\
            &= (cf)(-s) && \text{\text{Definition of scalar multiplication in \(\mathcal{F}(\mathbb{R}, \mathbb{R})\)}}
\end{align*}
Now we proceed to show that \(\mathcal{F}(\mathbb{R}, \mathbb{R})\) is a vector space over the reals.

\subsubsection*{(VS 1)}
Let \(f, g \in \mathcal{F}(\mathbb{R}, \mathbb{R})\) and \( s \in \mathbb{R}\):

\begin{align*}
    (f + g)(s)  &=  f(s) + g(s)             &&     \text{Definition of addition in \(\mathcal{F}(\mathbb{R}, \mathbb{R})\)} \\[6pt] 
                &=  g(s) + f(s)             &&     \text{Commutativity of \(\mathbb{R}\)} \\[6pt] 
                &=  (g + f)(s)              &&     \text{Definition of addition in \(\mathcal{F}(\mathbb{R}, \mathbb{R})\)}
\end{align*}

\subsubsection*{(VS 2)}

Let \(f, g, h \in \mathcal{F}(\mathbb{R}, \mathbb{R})\) and \( s \in \mathbb{R}\):

\begin{align*}
    \Big((f+g) + h\Big)\Big(s\Big) &= (f + g)(s) + h(s) && \text{Definition of addition in \(\mathcal{F}(\mathbb{R}, \mathbb{R})\)} \\
                                   &= \Big(f(s) + g(s)\Big) + h(s) && \text{Definition of addition in \(\mathcal{F}(\mathbb{R}, \mathbb{R})\)} \\
                                   &= f(s) + \Big(g(s) + h(s)\Big) && \text{Associativity of addition in \(\mathbb{R}\)} \\
                                   &= f(s) + \Big((g+h)(s)\Big) && \text{Definition of addition in \(\mathcal{F}(\mathbb{R}, \mathbb{R})\)} \\
                                   &= \Big(f + (g + h)\Big)\Big(s\Big) && \text{Definition of addition in \(\mathcal{F}(\mathbb{R}, \mathbb{R})\)}
\end{align*}

\subsubsection*{(VS 3)}

Let \(f\in \mathcal{F}(\mathbb{R}, \mathbb{R})\) and \( s \in \mathbb{R}\). Let also \(g(s) = 0\) for every \(s \in \mathbb{R}\).
We first want to make a point that \(g\) is an even function. Since \( s \in \mathbb{R}\), it must have an additive inverse denoted \(-s\) which is
also in \(\mathbb{R}\). Then:

\begin{equation*}
    g(s) = 0 = g(-s)
\end{equation*}
because \(g\) takes zero for any argument. We can prove now (VS 3):

\begin{align*}
    f(s) + g(s) = f(s) + 0 =  f(s) \ \  \text{Existence of additive identity in \(\mathbb{R}\)}
\end{align*}

\subsubsection*{(VS 4)}

Let \(f\in \mathcal{F}(\mathbb{R}, \mathbb{R})\) and \( s \in \mathbb{R}\). We have:

\begin{align*}
    \Big(f + (-f)\Big)(s) &= f(s) + \Big(-f(s)\Big) && \text{Definition of addition in \(\mathcal{F}(\mathbb{R}, \mathbb{R})\)} \\
    &= f(s) - f(s) &&  \text{Definition of addition in \(\mathbb{R}\)}  \\
    &= 0 && \text{Existence of additive inverse in \(\mathbb{R}\)}
\end{align*}


\subsubsection*{(VS 5)}

Let \(f\in \mathcal{F}(\mathbb{R}, \mathbb{R})\), \( s \in \mathbb{R}\). Then:

\begin{equation*}
    (1f)(s) = 1[f(s)] \ \ \text{\text{Definition of scalar multiplication in \(\mathcal{F}(\mathbb{R}, \mathbb{R})\)}}
\end{equation*}
\begin{equation*}
    = f(s) \ \ \text{Existence of multiplicative identity in \(\mathbb{R}\)}
\end{equation*}

\subsubsection*{(VS 6)}

Let \(f\in \mathcal{F}(\mathbb{R}, \mathbb{R})\) and \( a, b, s \in \mathbb{R}\). We have:

\begin{align*}
    \Big((ab)f\Big)\Big(s\Big) &= (ab)\Big[f(s)\Big] && \text{\text{Definition of scalar multiplication in \(\mathcal{F}(\mathbb{R}, \mathbb{R})\)}}\\
    &= a\Big(b\Big[f(s)\Big]\Big) && \text{Associative law for multiplication in \(\mathbb{R}\)} \\
    &= a\Big[(bf)(s)\Big] && \text{\text{Definition of scalar multiplication in \(\mathcal{F}(\mathbb{R}, \mathbb{R})\)}} \\
    &= \Big(a(bf)\Big)\Big(s\Big) &&  \text{\text{Definition of scalar multiplication in \(\mathcal{F}(\mathbb{R}, \mathbb{R})\)}} 
\end{align*}

\subsubsection*{(VS 7)}

Let \(f, g \in \mathcal{F}(\mathbb{R}, \mathbb{R})\) and \( a, s \in \mathbb{R}\):

\begin{align*}
    \Big(a(f+g)\Big)\Big(s\Big) &= a\Big[(f+g)(s)\Big] && \text{\text{Def. of scalar multipli. in \(\mathcal{F}(\mathbb{R}, \mathbb{R})\)}}\\
    &= a\Big[f(s) + g(s)\Big]  && \text{Def. of addition in \(\mathcal{F}(\mathbb{R}, \mathbb{R})\)} \\
    &= a\Big[f(s)\Big] + a\Big[g(s)\Big] && \text{Distributive law in \(\mathbb{R}\)} \\ 
    &= \Big((af)\Big)\Big(s\Big) + \Big((ag)\Big)\Big(s\Big) && \text{\text{Def. of scalar multipli. in \(\mathcal{F}(\mathbb{R}, \mathbb{R})\)}}
\end{align*}

\subsubsection*{(VS 8)}

Let \(f \in \mathcal{F}(\mathbb{R}, \mathbb{R})\) and \( a, b, s \in \mathbb{R}\):

\begin{align*}
    \Big((a + b)f\Big)\Big(s\Big) &= (a+b)\Big[f(s)\Big] && \text{\text{Def. of scalar multipli. in \(\mathcal{F}(\mathbb{R}, \mathbb{R})\)}}\\
    &= a\Big[f(s)\Big] + b\Big[f(s)\Big] && \text{Distributive law in \(\mathbb{R}\)} \\ 
    &= (af)(s) + (bf)(s) && \text{\text{Def. of scalar multipli. in \(\mathcal{F}(\mathbb{R}, \mathbb{R})\)}}\\
\end{align*}
