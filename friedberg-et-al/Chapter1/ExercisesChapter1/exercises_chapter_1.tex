\section*{Exercise 1.2.8}

Suppose \(V\) is a vector space over a field \(F\). Assume \(x, y \in V\) and \(a, b \in F\). From 
the properties of linear spaces:

\begin{equation*}
    (a + b)(x + y) = (a + b)x + (a + b)y \ \ \ \emph{Using  VS 7}
\end{equation*}
\begin{equation*}
    = ax + bx + ay + by \ \ \ \emph{Using VS 8}
\end{equation*}
\begin{equation*}
    = ax + ay + bx + by \ \ \ \emph{Using VS 1}
\end{equation*}

and this completes the proof.

\section*{Exercise 1.2.9}

\begin{theorem}
    Corollary 1
\end{theorem}
\begin{proof}
    Assume we are working in a vector space \(V\) over a field \(F\). Assume for contradiction,
that \(0 \in V\) is not unique. Then there exists two zero vectors \(0_{1} \neq 0_{2}\).
It follows by (VS 3) and (VS 1):

\begin{equation*}
    0_{1} = 0_{1} + 0_{2} = 0_{2} + 0_{1} = 0_{2}
\end{equation*}

but \(0_{1} \neq 0_{2}\), contradiction.
\end{proof}

\begin{theorem}
    Corollary 2
\end{theorem}
\begin{proof}
    Assume a vector space \(V\) over a field \(F\). Suppose \(x \in V\) and assume for contradiction
    that there exists two vectors in \(V\), \(y_1 \neq y_2\) such that:
    \begin{equation*}
        x + y_1 = 0 \ \ \mbox{and} \ \ x + y_2 = 0
    \end{equation*}
    It follows then:
    \begin{equation*}
        x + y_1 = 0 = x + y_2
    \end{equation*}
    And using theorem 1.2.1 (Cancellation Law For Vector Addition):
    \begin{equation*}
        y_1 = y_2
    \end{equation*}
    Contradiction.
\end{proof}

\begin{theorem}
    Theorem 1.2 c
\end{theorem}
\begin{proof}
    Assume a vector space \(V\) over a field \(F\). Suppose \(a \in F\) and \(0\) being a zero vector in \(V\).
    We get:

    \begin{equation*}
        a0 + a0 = a(0 + 0) = a0
    \end{equation*}
    adding \(-a0\) to both sides and using (VS 4) and (VS 3):
    \begin{equation*}
        a0 + (a0 + (-a0)) = a0 - a0 = 0
    \end{equation*}
    gives us:
    \begin{equation*}
        a0 = 0
    \end{equation*}
    and this completes the proof.
\end{proof}
