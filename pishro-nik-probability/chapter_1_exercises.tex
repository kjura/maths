\section*{Problem 8*}

Niech \(x \in A\). Istnieje wtedy \(n_0 \in \mathbb{N}\) takie, że \(0 \leq x < 1 - \frac{1}{n_0} < 1\). Stąd \(x \in [0, 1)\).
W drugą stronę, niech \(x \in [0, 1)\). Wiadomym jest, że \(\lim_{n \to \infty} 1 - \frac{1}{n} = 1\). Zauważmy, że
\( \frac{1 - x}{2} > 0 \). Z definicji granicy ciągu istnieje zatem \(n_0 \in \mathbb{N}\) takie, że:
\begin{equation*}
    |\big(1 - \frac{1}{n}\big) - 1| = \frac{1}{n} < \frac{1 - x}{2} \ \ \mbox{dla} \ \ n \geq n_0
\end{equation*}
Stąd:
\begin{equation*}
    0 \leq x < 1 - \frac{2}{n} < 1 - \frac{1}{n} < 1 \ \ \mbox{dla} \ \ n \geq n_0
\end{equation*}
i stwierdzamy, że \(x \in A\).
\subsection*{Appendix}
W drugą stronę (bez analizy). Szukamy \(n \in \mathbb{N}\) takiego, że \(x < 1 - \frac{1}{n}\). Mamy zatem:

\begin{equation*}
    x < 1 - \frac{1}{n} \iff \frac{1}{n}  < 1 - x \iff \frac{1}{1 - x} < n 
\end{equation*}
Zatem dla \(x \in [0, 1)\) istnieje \(n > \frac{1}{1 - x}\) gdzie \(x \in A_{n}\). Stąd \(x \in A\).

\section*{Problem 9*}

First, observe that \( 0 < \frac{1}{n} \) for every \(n \geq 1\) and \(0\) is in every \(A_n\). Then \( 0 \in A \).
\newline
It is known that \(A \subseteq [0, \infty)\). Indeed, if \(a \in A\) then:
\begin{equation*}
    (\forall n \geq 1) \ \ a \in A_{n} \implies a \in A_{1} = [0, 1) \implies a \in [0, 1] \implies a \in [0, \infty)
\end{equation*}

Assume now that \(a > 0\). We know that \(\lim_{n \to \infty} \frac{1}{n} = 0\).
Let \(\epsilon = a\).  From the definition of the limit there exists \(n_0 \in \mathbb{N}\) such that:

\begin{equation*}
    a = \epsilon > |\frac{1}{n} - 0| = |\frac{1}{n}| = \frac{1}{n} \ \ \mbox{dla} \ \ n \geq n_0
\end{equation*}

Then:

\begin{equation*}
    a > \frac{1}{n} > 0 \ \ \mbox{dla} \ \ n \geq n_0
\end{equation*}

We have \(a \notin A_{n_0}\). Using contraposition:

\begin{equation*}
    \Big( a \notin \{0\} \implies a \notin A \Big) \iff \Big( a \in A \implies a \in \{0\} \Big)
\end{equation*}

and finally we have:

\begin{equation*}
    A = \bigcap_{n=1}^{\infty} [0, \frac{1}{n}) = \{ 0 \}
\end{equation*}

\section*{Problem 16}
\subsection*{a.}

The sample space \(S = \{1, 2, 3, \dots\}\) is a countable set. Since \(P(S) = 1\)
by the third axiom of probability we can write:

\begin{equation*}
    P(S) = P(\{1, 2, 3, \dots\}) = P(\bigcup_{s_{j} \in S} \{s_j\}) = \sum_{s_{j} \in S}P(s_{j}) = 1
\end{equation*}
Next, we can use natural indices to represent above:
\begin{equation*}
    \sum_{s_{j} \in S}P(s_{j}) = \sum_{k=1}P(k)  = \sum_{k=1} \frac{c}{3^{k}} = c \sum_{k=1} \Big(\frac{1}{3}\Big)^{k} = 1
\end{equation*}

We seek:
\begin{equation*}
    c = \frac{1}{\sum_{k=1} \Big(\frac{1}{3}\Big)^{k}}
\end{equation*}

We quickly recognize that \(\sum_{k=1} \frac{1}{3^{k}}\) is a geometric series. Using the series sum:
\begin{equation*}
    \sum_{k=1} \Big(\frac{1}{3}\Big)^{k} = \Big( 1 + \frac{1}{3} + \Big(\frac{1}{3}\Big)^{2} + \Big(\frac{1}{3}\Big)^{3} + \dots \Big) - 1
\end{equation*}
\begin{equation*}
    = \Big(\frac{1}{1 - \frac{1}{3}}\Big) - 1 = \Big(\frac{1}{\frac{2}{3}}\Big) - 1 = \frac{3}{2} - 1 = \frac{1}{2}
\end{equation*}
So finally:
\begin{equation*}
    c = \frac{1}{\frac{1}{2}} = 2
\end{equation*}

\subsection*{b.}
Since in the previous step we have found \(c\), we now know the full formula for the probability
of any event of our sample space, i.e:
\begin{equation*}
    P(k) = \frac{2}{3^{k}}
\end{equation*}
We write:
\begin{equation*}
    P(\{2, 4, 6\}) = \sum_{k \in \{2, 4, 6\}}P(k) = P(2) + P(4) + P(6) =
\end{equation*}
\begin{equation*}
    = \frac{2}{3^{2}} + \frac{2}{3^{4}} + \frac{2}{3^{6}} = \frac{2}{9} + \frac{2}{81} + \frac{2}{729} = 
\end{equation*}
\begin{equation*}
    = \frac{182}{729} \approx 0.25
\end{equation*}

\subsection*{c.}

We want to find \(P(\{3, 4, 5, \dots\})\). We have:

\begin{equation*}
    P(\{3, 4, 5, \dots\}) = 2 \sum_{k=3}^{\infty} \frac{1}{3^{k}} = 
\end{equation*}
\begin{equation*}

\section*{Problem 17}

Let \(\Omega\) be a sample space indicating winnings of the teams involved i.e
\begin{equation*}
    \Omega = \{A, B, C, D\}
\end{equation*}
Since this is a discrete model, we can sum the probability of individual elements in the set:

\begin{equation*}
    1 = P(\Omega) = P(\{A, B, C, D\}) = P(\{A\} \cup \{B\} \cup \{C\} \cup \{D\})
\end{equation*}
\begin{equation*}
    = P(A) + P(B) + P(C) + P(D)
\end{equation*}

We know that:
\begin{equation*}
    P(A) = P(B), P(C) = 2P(D), P(A \cup C) = 0.6
\end{equation*}
We have \(P(A \cup C) = P(A) + P(C) = 0.6\). So \(P(A) = 0.6 - P(C)\). We find \(P(C)\):
\begin{equation*}
    P(C) = 1 - P(A) - P(B) - P(D)  = 1 - 2P(A) - \frac{P(C)}{2} = 1 - 2(0.6 - P(C)) - \frac{P(C)}{2} 
\end{equation*}
\begin{equation*}
    = 1 - 1.2 + 2P(C) - \frac{P(C)}{2} = -0.2 + \frac{3P(C)}{2}
\end{equation*}
Then:
\begin{equation*}
    P(C) = -0.2 + \frac{3P(C)}{2}
\end{equation*}
\begin{equation*}
    P(C) - \frac{3P(C)}{2} = -0.2
\end{equation*}
\begin{equation*}
    -\frac{P(C)}{2} = -0.2
\end{equation*}
\begin{equation*}
    P(C) = 0.4
\end{equation*}
From above, \(P(A) = 0.6 - P(C) = 0.6 - 0.4 = 0.2\). So \(P(A) = P(B) = 0.2\). Finally \(P(D) = \frac{P(C)}{2} = \frac{0.4}{2} = 0.2\). Summarizing:
\begin{equation*}
    P(A) = 0.2, P(B) = 0.2, P(C) = 0.4, P(D) = 0.2
\end{equation*}
