\subsection*{a.}
The range, \(R_{X}\) is:
\begin{equation*}
    R_{X} = \{0, 1, 2\}
\end{equation*}
\subsection*{b.}
\begin{equation*}
    P(X \geq 1.5) = 1 - P(X < 1.5) = 1 - \sum_{x \in \{0, 1\}}P_{X}(x)
\end{equation*}
\begin{equation*}
    = 1 - \Big(P(X=0) + P(X=1)\Big) = 1 - \frac{1}{2} + \frac{1}{3} = \frac{5}{6}
\end{equation*}

\begin{equation*}
    P(0 < X < 2) = P(X \in (0, 2)) = P(X \in \{1\}) = P(X = 1) = \frac{1}{3}
\end{equation*}
